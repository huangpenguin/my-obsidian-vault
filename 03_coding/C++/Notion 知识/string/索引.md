# string

```cpp
//修改所有元素
//因为这里其实是要更改s，所以循环控制变量c要设置成引用，被依次绑定到序列每个元素上
int main(){
    string s="eosifhoeh5go5ehoigho23fhu459g6";
    for (auto &c:s)
    c='x';
    cout<<s<<endl;
}
```

```cpp
//读入一个字符串，删除其中的空格和标点符号
#include <iostream>
#include <sstream>
using namespace std;

int main() {
    string s = "eosifhoe,.h   5go5ehoi,.,.,.   gho23,.  fh  u4,.,.5  9g6";
    stringstream ss;

    for (auto& c : s) {
        if (!std::ispunct(c) && !std::isspace(c)) {
            ss << c; //比普通的+=性能好
        }
    }
    std::cout << ss.str() << std::endl;
    return 0;
}
```

#include <iostream>
#include <string>
#include <regex>
int main()
{
std::string line, pattern;
std::cout << "文字列> ";
std::getline(std::cin, line);
std::cout << "正規表現> ";
std::getline(std::cin, pattern);
std::regex re{pattern};
std::smatch match;
if (std::regex_search(line, match, re))
{
for (std::size_t i = 0; i < match.size(); ++i)
{
std::cout << match.str(i) << std::endl;
}
}
else
{
std::cout << "正規表現にマッチしませんでした" << std::endl;
}
}
