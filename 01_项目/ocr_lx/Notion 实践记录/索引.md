# ----- 新增提取逻辑示例 -----

def generate_component_info_results(ocr_result=None) -> list["ExtractionResult"]:
    """部件信息の結果を生成 - 新しい提取逻辑示例"""
    if ocr_result is None:
        # OCRデータがない場合はダミーデータ
        room = generate_room_numbers()
        return [
            _make_result(
                room=room,
                item="部件信息",
                index="COMPONENT_INFO",
                bath_product_number="",
                importance="MEDIUM",
                extracted_value="null",
                reference_pdf_type="製作依頼書",
            )
        ]
    
    # 实际的提取逻辑
    results = []
    room = generate_room_numbers()
    
    # 这里添加你的具体提取逻辑
    # 比如: 查找特定的表格、文本模式等
    
    # 示例: 简单的文本搜索
    component_found = False
    for img_idx, image in enumerate(ocr_result.images):
        if image.text_lines:
            for line in image.text_lines:
                if "部件" in line.text or "コンポーネント" in line.text:
                    component_found = True
                    break
    
    result = _make_result(
        room=room,
        item="部件信息",
        index="COMPONENT_INFO",
        bath_product_number="",
        importance="HIGH" if component_found else "LOW",
        extracted_value="found" if component_found else "not_found",
        reference_pdf_type="製作依頼書",
    )
    results.append(result)
    
    return results
