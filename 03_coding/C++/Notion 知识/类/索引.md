# 类

```cpp
#include <iostream>
#include <string>

class Product {
private:
	int id;
	int price;
	int stock;

public:
	struct Result {
		int id;
		int price;
		int stock;
	};

	Product(int v1, int v2, int v3)
		: id(v1), price(v2), stock(v3) {}

	Result getValues() const {//此处的cosnt说明这里是常量成员函数，不会改变值
		return { id, price, stock };
	}
};

int main() {
	Product ex(0, 140, 300);
	Product::Result result = ex.getValues();
	std::cout << "id: " << result.id << ", price: " << result.price << ", stock: " << result.stock << std::endl;
	return 0;
}
```
