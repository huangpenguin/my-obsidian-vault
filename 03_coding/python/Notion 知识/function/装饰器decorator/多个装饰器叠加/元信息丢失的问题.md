# 元信息丢失的问题

```python
def log_start_end(func):
    def wrapper(*args, **kwargs):
        print(f"Starting '{func.__name__}'")
        result = func(*args, **kwargs)
        print(f"Finished '{func.__name__}'")
        return result
    return wrapper

def log_arguments(func):
    def wrapper(*args, **kwargs):
        print(f"Arguments for '{func.__name__}': args={args}, kwargs={kwargs}")
        return func(*args, **kwargs)
    return wrapper

@log_start_end
@log_arguments
def my_range(first=0, last=10, step=1):
    number = first
    while number < last:
        yield number
        number += step

# 使用装饰器的生成器
print("First run:")
for x in my_range(1, 5):
    print(x)

```

结果

First run:
Starting 'wrapper'
Arguments for 'my_range': args=(1, 5), kwargs={}
Finished 'wrapper'
1
2
3
4

上面的代码出现了元信息丢失的问题

print(f"Starting '{func_para.__name__}'")预测应该是打印log_arguments的名字，但是log_start_end中的元信息发生了丢失所以无法正确打印。

参见[https://python3-cookbook.readthedocs.io/zh-cn/latest/c09/p02_preserve_function_metadata_when_write_decorators.html](https://python3-cookbook.readthedocs.io/zh-cn/latest/c09/p02_preserve_function_metadata_when_write_decorators.html)

所以需要使用wraps

```python
from functools import wraps

def log_start_end(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"Starting '{func.__name__}'")
        result = func(*args, **kwargs)
        print(f"Finished '{func.__name__}'")
        return result
    return wrapper

def log_arguments(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"Arguments for '{func.__name__}': args={args}, kwargs={kwargs}")
        return func(*args, **kwargs)
    return wrapper

@log_start_end
@log_arguments
def my_range(first=0, last=10, step=1):
    number = first
    while number < last:
        yield number
        number += step

print("First run:")
for x in my_range(1, 5):
    print(x)

```
